    SPOTS_EVENTS {
        string id PK
        string title "Name of destination / event"
        string category "Culture|Heritage|Eco Tourism|Food & Crafts"

        float longitude "GPS decimal coordinates"
        date event_date "Date of festival or schedule"
        text description "Spot overview & tourist guidelines"

        string status "Approved|Pending Admin Approval|Rejected"

        int rating "1 to 5 Star numerical rating"
        text comment "Visitor testimonial or inquiry"

        string replied_by FK "Staff username who replied"

Diagram 4: System Flowchart

End-to-end procedural flowchart showing decision logic, progressive lockout counter, role branching, and data store writes.

Loading System Flowchart...
flowchart TD
    Start(["User Opens TourLynk"]) --> CheckSession{"Valid Active Session?"}
    CheckSession -- "Yes" --> CheckInactivity{"Inactive > 15 Mins?"}
    CheckInactivity -- "Yes" --> AutoLogout["Trigger Auto-Logout & Record Log"] --> LoginPage["Show Login Screen"]
    CheckInactivity -- "No" --> RouteRole["Route to Role Dashboard"]

    CheckSession -- "No" --> LoginPage
    LoginPage --> AuthChoice{"User Action?"}

    AuthChoice -- "Forgot Password" --> OtpRecovery["Enter Email -> Generate 6-Digit OTP"]
    OtpRecovery --> CheckOtp{"Valid OTP & < 3 mins?"}
    CheckOtp -- "Yes" --> ResetPwd["Reset Password with Rules"] --> LoginPage
    CheckOtp -- "No" --> InvalidateOtp["Increment OTP Failures"] --> LoginPage

    AuthChoice -- "Register Account" --> RegInput["Enter Name, Email, Password"]
    RegInput --> CheckPwd{"Meets IAS Complexity Rules?"}
    CheckPwd -- "Yes" --> SaveUser["Hash Password & Create Account"] --> LoginPage
    CheckPwd -- "No" --> ShowPwdErrors["Show Dynamic Rule Violations"] --> RegInput

    AuthChoice -- "Login" --> VerifyCreds{"Credentials Valid?"}
    VerifyCreds -- "No" --> IncFailures["Increment Failed Attempts"]
    IncFailures --> Check3Failures{"Attempts >= 3?"}
    Check3Failures -- "Yes" --> Lockout["Apply Progressive Lockout: 1m / 5m / 15m & CAPTCHA"] --> LoginPage
    Check3Failures -- "No" --> LoginPage

    VerifyCreds -- "Yes" --> ResetFailCount["Reset Failed Attempts to 0"]
    ResetFailCount --> CreateSession["Issue Session & Start Inactivity Watcher"]
    CreateSession --> RouteRole

    RouteRole -- "Visitor" --> TouristHub["Browse Spots & GPS Distance"]
    RouteRole -- "Staff" --> StaffHub["Review Passes & Propose Places"]
    RouteRole -- "Admin" --> AdminHub["User CRUD, Audit Logs, Backup"]
